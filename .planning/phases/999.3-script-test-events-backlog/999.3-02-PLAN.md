function resolveScriptParameters(ctx, scriptId: string): Array<{key:string;value:string}> | undefined {
    const cached = ctx.workspaceState.get<Record<string, unknown>>('altium365.scriptParams.' + scriptId);
    if (!cached || typeof cached !== 'object') return undefined;
    const out: Array<{key:string;value:string}> = [];

        if (value === undefined || value === null) continue;
        out.push({ key, value: String(value) });

}
