<objective>
Ship the editor surface for test events. Adds the writable virtual document scheme `altium365-event://` (backed by a FileSystemProvider that round-trips through globalState) and the declarative JSONSchema binding. After this plan, opening `altium365-event://<identity>/<name>.json` via `vscode.workspace.openTextDocument` produces an editable JSON buffer with autocomplete-style schema validation, and Cmd+S commits the change to globalState.

Purpose: Reuses VS Code's native JSON editor (syntax, formatting, schema squigglies) for zero cost. Replaces what would otherwise be a custom WebView form. RESEARCH §Q1 verified this is the cleanest approach vs registerTextDocumentContentProvider (read-only) or untitled: scheme (wrong Save-As semantics).

Output: One new module + activation wiring edit + one package.json contribution. Two commits.

       const SCHEMA_BASENAME = 'schemas/test-event.schema.json';
       function parseEventUri(uri: vscode.Uri): { identity: string; eventName: string } | undefined {
           if (uri.scheme !== SCHEME) return undefined;
           const identity = decodeURIComponent(uri.authority || '');
           const raw = uri.path.startsWith('/') ? uri.path.slice(1) : uri.path;
           if (!raw.toLowerCase().endsWith('.json')) return undefined;
           const eventName = decodeURIComponent(raw.slice(0, raw.length - '.json'.length));
           if (!identity || !eventName) return undefined;
           return { identity, eventName };
       }
       export function buildEventUri(identity: string, eventName: string): vscode.Uri {
           return vscode.Uri.parse(`${SCHEME}://${encodeURIComponent(identity)}/${encodeURIComponent(eventName)}.json`);
       }
