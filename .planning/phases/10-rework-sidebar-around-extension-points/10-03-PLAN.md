Add a helper function at the top of the file (after imports, before `registerScriptCommands`):

```typescript
/**
 * Extract script metadata from either a script node or a script assignment node.
 * Returns undefined if the node is neither, or if it's a non-script assignment.
 */
function extractScriptContext(
    node: any
): { scriptId: string; scriptName: string; workspaceAuthId: string; workspaceUrl: string } | undefined {
    if (node.kind === 'script') {
        return {
            scriptId: node.script.scriptId,
            scriptName: node.script.name,
            workspaceAuthId: node.workspaceAuthId,
            workspaceUrl: node.workspaceUrl,
        };
    }
    
    if (node.kind === 'assignmentNode') {
        if (node.assignment.type !== 'GloCusScriptAssignment') {
            return undefined;
        }
        
        return {
            scriptId: node.assignment.scriptId || '',
            scriptName: node.assignment.name,
            workspaceAuthId: node.workspaceAuthId,
            workspaceUrl: node.workspaceUrl,
        };
    }
    
    return undefined;
}
```

In each of the four command handlers, replace the script-node-only extraction with the helper:

```typescript
const scriptName = node.script.name;
```

**After:**

```typescript
const scriptContext = extractScriptContext(node);
if (!scriptContext) {
    vscode.window.showErrorMessage('This command works only on scripts or script assignments.');

}
const { scriptId, scriptName, workspaceAuthId, workspaceUrl } = scriptContext;
```
