```ts
export interface ExecutionResult {
    scriptExecutionId: string;
    status: string;             // String! in schema — see Pitfall 5
    exitCode: number | null;
    startedAt: string | null;
    completedAt: string | null;
    failureReason: string | null;
}

export interface ExecutionLogPage {
    logs: string[];
    nextToken: string;          // String! per schema; empty string when no more
}

export interface ExecuteScriptInput {
    scriptId: string;
    scriptVersionId?: string;                // optional; omit to use latest
    parameters?: Array<{ key: string; value: string }>;
}

export async function executeScript(
    endpoint: string,
    workspaceToken: string,
    input: ExecuteScriptInput
): Promise<{ scriptExecutionId: string; status: string }>;

export async function getExecutionResult(
    endpoint: string,
    workspaceToken: string,
    scriptExecutionId: string
): Promise<ExecutionResult>;

export async function getExecutionLogs(
    endpoint: string,
    workspaceToken: string,
    scriptExecutionId: string,
    limit: number,
    nextToken: string | null
): Promise<ExecutionLogPage>;
```

GraphQL strings (verbatim from RESEARCH §Code Examples + introspection):

```graphql
mutation ExecuteScript($input: GloScrExecuteScriptInput!) {
  gloScrExecuteScript(input: $input) {
    gloScrScriptExecution { scriptExecutionId status }
  }
}

query GetExecResult($id: String!) {
  gloScrScriptExecutionResult(scriptExecutionId: $id) {
    scriptExecutionId
    status
    exitCode
    startedAt
    completedAt
    failureReason
  }
}

query GetExecLogs($id: String!, $limit: Int!, $nextToken: String) {
  gloScrScriptExecutionResult(scriptExecutionId: $id) {
    logs(limit: $limit, nextToken: $nextToken) { logs nextToken }
  }
}
```

Note: status + logs are split into two queries (rather than the single combined query in RESEARCH §Pattern 2) so the two responses can be sized independently and a malformed log-page can't poison the status read. Both run in parallel each tick via `Promise.all`.

    Add a private (module-scope, non-exported) async function `runPollLoop` that drives the streaming. Signature:

    ```ts
    async function runPollLoop(
        apiUrl: string,
        wsToken: string,
        execId: string,
        args: ExecuteRemoteArgs,
        progress: vscode.Progress<{ message?: string; increment?: number }>,
        token: vscode.CancellationToken
    ): Promise<void>;
    ```

    Constants near the function:
    ```ts
    const POLL_INTERVAL_MS = 1500;                       // RESEARCH recommended
    const LOG_PAGE_LIMIT = 500;                          // RESEARCH §Code Examples
    const MAX_WALLCLOCK_MS = 10 * 60 * 1000;             // 10 min backstop (Pitfall 5)
    const TERMINAL_STATUSES = new Set([                  // case-folded
        'succeeded', 'failed', 'cancelled', 'stopped', 'completed', 'error'
    ]);
    ```

    Loop body:

    1. `let nextToken: string | null = null;`
    2. `let cancelled = false; token.onCancellationRequested(() => { cancelled = true; });` — D-10: client-side abandon only; no server cancel exists (RESEARCH §Architecture decisions).
    3. `const startedAt = Date.now();`
    4. `let lastStatus = '';`
    5. `while (!cancelled)`:
       - Wall-clock check: `if (Date.now() - startedAt > MAX_WALLCLOCK_MS) { args.output.appendLine('[Altium 365] Remote execution backstop reached (10 min) — abandoning poll.'); break; }`
       - Fetch status + logs **in parallel** via `Promise.all([getExecutionResult(...), getExecutionLogs(apiUrl, wsToken, execId, LOG_PAGE_LIMIT, nextToken)])`. Wrap in try/catch — on transient error, `args.output.appendLine('[Altium 365] poll tick failed (will retry): ' + (e as Error).message);` then `await sleep(POLL_INTERVAL_MS); continue;`.
       - For every line in `logPage.logs`, `args.output.appendLine(line)` (one append per line so VS Code's renderer streams them).
       - `if (logPage.nextToken) { nextToken = logPage.nextToken; }` — never go backwards; empty-string means "no advance" (RESEARCH says `String!` non-null — empty == no new page).
       - `if (result.status !== lastStatus) { progress.report({ message: result.status }); lastStatus = result.status; args.output.appendLine('[Altium 365] status=' + result.status); }`
       - `if (TERMINAL_STATUSES.has(result.status.toLowerCase())) { /* terminal — write footer and return */ writeFooter(args.output, result); return; }`
       - `await sleep(POLL_INTERVAL_MS);`
    6. Outside the loop (cancellation path):
       - `args.output.appendLine('[Altium 365] Remote execution cancelled (server-side execution continues)');` (D-10 truth).

    Helpers (same file, non-exported):
    ```ts
    function sleep(ms: number): Promise<void> { return new Promise(r => setTimeout(r, ms)); }

    function writeFooter(out: vscode.OutputChannel, r: ExecutionResult): void {
        const exit = r.exitCode === null || r.exitCode === undefined ? 'ok' : String(r.exitCode);
        out.appendLine('[Altium 365] Remote execution finished (status=' + r.status + ', exit=' + exit + ')');
        if (r.failureReason) { out.appendLine('[Altium 365] failureReason: ' + r.failureReason); }
    }
    ```

    Notes for executor:
    - The two GraphQL queries on each tick are run in parallel via `Promise.all` to keep each tick close to 1.5 s end-to-end (network-bound). RESEARCH §Pattern 2 used a single combined query; we split them for response-size independence (see `<interfaces>` rationale) — verify by inspecting the OutputChannel timestamps during UAT.
    - Logs append BEFORE the terminal-check so that the final log batch is rendered even on the same tick the status flips to terminal.
    - Curated terminal set is permissive on purpose (Pitfall 5). Every observed `status` value gets logged via the `status=...` line, so UAT can catch a missing terminal value and we fix-forward by adding to the set.
    - Do NOT close `args.output`; it is the module-level singleton owned by `extension.ts` (D-09).
